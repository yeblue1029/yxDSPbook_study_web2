寄存器配置代码

// McBSP初始化配置
void InitMcbsp(void)
{
    // 停止McBSP
    McbspaRegs.SPCR2.all = 0x0000;
    McbspaRegs.SPCR1.all = 0x0000;
    
    // 配置接收控制寄存器 (RCR)
    McbspaRegs.RCR2.all = 0x000000;
    McbspaRegs.RCR1.bit.RWDLEN1 = 2;  // 16位数据
    
    // 配置发送控制寄存器 (XCR)
    McbspaRegs.XCR2.all = 0x000000;
    McbspaRegs.XCR1.bit.XWDLEN1 = 2;  // 16位数据
    
    // 配置采样率生成器 (SRGR)
    McbspaRegs.SRGR2.all = 0x000000;
    McbspaRegs.SRGR2.bit.CLKSM = 1;   // LSPCLK作为时钟源
    McbspaRegs.SRGR2.bit.FPER = 159;  // 帧周期
    McbspaRegs.SRGR1.bit.CLKGDV = 7;  // 分频系数
    
    // 配置引脚控制寄存器 (PCR)
    McbspaRegs.PCR.all = 0x000000;
    McbspaRegs.PCR.bit.FSXM = 1;       // 主机提供帧同步
    McbspaRegs.PCR.bit.FSXP = 0;       // 帧同步高有效
    McbspaRegs.PCR.bit.CLKXM = 1;     // 主机提供时钟
    McbspaRegs.PCR.bit.CLKXP = 0;      // 上升沿采样
    
    // 使能McBSP
    McbspaRegs.SPCR2.all = 0x0001;
    McbspaRegs.SPCR1.all = 0x0001;
}
