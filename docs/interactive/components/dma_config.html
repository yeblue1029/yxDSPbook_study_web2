寄存器配置代码

// DMA初始化配置
void InitDMA(void)
{
    // 停止DMA
    DMACtrlRegs.DEBUGCTRL.bit.SOFTRESET = 1;
    
    // 配置CH1通道
    // 源地址
    DmaRegs.CH1.SRC_ADDR.all = 0x00007000;
    // 目标地址
    DmaRegs.CH1.DST_ADDR.all = 0x00008000;
    // 突发大小: 16字
    DmaRegs.CH1.BURST_SIZE.all = 16;
    // 传输大小: 8次
    DmaRegs.CH1.TRANSFER_SIZE.all = 8;
    
    // 配置MODE寄存器
    DmaRegs.CH1.MODE.bit.PERINTSEL = 1;    // 触发源: ADC排序器1
    DmaRegs.CH1.MODE.bit.PERINTE = 1;       // 使能外设中断触发
    DmaRegs.CH1.MODE.bit.ONESHOT = 0;       // 关闭ONESHOT
    DmaRegs.CH1.MODE.bit.CHINTMODE = 0;     // 传输结束中断
    
    // 清除任何挂起的标志
    DmaRegs.CH1.CONTROL.bit.PERINTCLR = 1;
    
    // 使能CH1通道
    DmaRegs.CH1.CONTROL.bit.RUN = 1;
}
