寄存器配置代码

// CPU定时器初始化配置
void InitCpuTimer(void)
{
    // 停止定时器
    CpuTimer0Regs.TCR.bit.TSS = 1;
    
    // 设置预分频
    CpuTimer0Regs.TDR.all = 0;
    
    // 设置周期值
    CpuTimer0Regs.TIM.all = 0xFFFFFFFF;
    CpuTimer0Regs.PRD.all = 99999999;
    
    // 启动定时器
    CpuTimer0Regs.TCR.bit.TSS = 0;
}
