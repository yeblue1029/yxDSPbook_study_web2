// BLDC电机控制初始化
void InitBLDC(void)
{
    // 配置ePWM1,2,3用于BLDC控制
    EPwm1Regs.TBPRD = 3750;           // 20kHz PWM
    EPwm1Regs.TBPHS.half.TBPHS = 0;
    EPwm1Regs.CMPA.half.CMPA = 1875;  // 50%占空比
    EPwm1Regs.AQCTLA.bit.CAU = AQ_SET;
    EPwm1Regs.AQCTLA.bit.CAD = AQ_CLEAR;
    
    // Hall传感器GPIO配置
    GpioCtrlRegs.GPAPUD.bit.GPIO0 = 0;    // 上拉使能
    GpioCtrlRegs.GPAQSEL1.bit.GPIO0 = 2;  // 6次采样滤波
    GpioCtrlRegs.GPAMUX1.bit.GPIO0 = 0;   // GPIO功能
    GpioCtrlRegs.GPADIR.bit.GPIO0 = 0;    // 输入
    
    // 更多GPIO配置...
}

// 6步换相表
const uint16_t commutationTable[8] = {
    0,      // 000 无效
    0x0A,   // 001: U->V, W高阻
    0x09,   // 010: W->U, V高阻
    0x06,   // 011: V->W, U高阻
    0x05,   // 100: U->W, V高阻
    0x02,   // 101: W->V, U高阻
    0x01,   // 110: V->U, W高阻
    0       // 111 无效
};

// Hall传感器中断处理
interrupt void hallISR(void)
{
    uint16_t hallValue;
    hallValue = (GpioDataRegs.GPADAT.bit.GPIO0 << 2) |
                (GpioDataRegs.GPADAT.bit.GPIO1 << 1) |
                GpioDataRegs.GPADAT.bit.GPIO2;
    
    // 根据Hall值查表换相
    uint16_t pwmPattern = commutationTable[hallValue];
    
    // 更新PWM输出
    updatePWM(pwmPattern);
    
    // 计算转速
    calculateSpeed();
    
    // 清除中断标志
    PieCtrlRegs.PIEACK.all = PIEACK_GROUP3;
}
