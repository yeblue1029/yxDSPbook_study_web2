I2C寄存器配置代码

// I2C初始化配置
void InitI2C(void)
{
    // 禁用I2C模块
    I2caRegs.I2CMDR.all = 0x0000;
    
    // 配置时钟频率
    // I2C模块时钟 = LSPCLK / (ICCL + ICD)
    // 100kHz: ICCL = ICD = 60 (约)
    // 400kHz: ICCL = ICD = 15 (约)
    I2caRegs.I2CSCLL.all = 60;
    I2caRegs.I2CSCLH.all = 60;
    
    // 从机地址
    I2caRegs.I2COAR.all = 0x0050;
    
    // 从机模式自己的地址
    I2caRegs.I2COAR.all = 0x0050;
    
    // 使能I2C模块
    I2caRegs.I2CMDR.all = 0x0020;
}

// I2C发送START信号
void I2CStart(void)
{
    I2caRegs.I2CMDR.all = 0x0020;  // STP = 0, STT = 1, IRS = 1
    while(I2caRegs.I2CSTR.bit.SCD == 0);
}

// I2C发送STOP信号
void I2CStop(void)
{
    I2caRegs.I2CMDR.bit.STP = 1;
    while(I2caRegs.I2CSTR.bit.SCD == 0);
}

// I2C发送数据
void I2CSend(uint8_t data)
{
    I2caRegs.I2CDXR.all = data;
    while(I2caRegs.I2CSTR.bit.ARDY == 0);
}

// I2C接收数据
uint8_t I2CReceive(void)
{
    while(I2caRegs.I2CSTR.bit.RRDY == 0);
    return I2caRegs.I2CDRR.all;
}
