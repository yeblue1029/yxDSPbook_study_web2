寄存器配置代码

// eCAP初始化配置
void InitECap(void)
{
    // 禁用ECAP
    ECap1Regs.ECCTL2.bit.TSCTRSTOP = 0;
    
    // 配置捕获沿
    ECap1Regs.ECCTL1.bit.CAP1POL = 0;  // 上升沿
    ECap1Regs.ECCTL1.bit.CAP2POL = 0;  // 上升沿
    ECap1Regs.ECCTL1.bit.CAP3POL = 1;  // 下降沿
    ECap1Regs.ECCTL1.bit.CAP4POL = 0;  // 上升沿
    
    // 配置预分频
    ECap1Regs.ECCTL1.bit.PRESCALE = 0;  // 1分频
    
    // 使能捕获寄存器加载
    ECap1Regs.ECCTL1.bit.CAPLDEN = 1;
    
    // 配置连续/单次模式
    ECap1Regs.ECCTL2.bit.CONT/ONESHT = 0;  // 连续模式
    
    // 启动计数器
    ECap1Regs.ECCTL2.bit.TSCTRSTOP = 1;
}
