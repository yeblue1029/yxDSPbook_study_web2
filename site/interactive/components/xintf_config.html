// XINTF初始化配置
void InitXintf(void)
{
    // 关闭所有zone访问
    XintfRegs.XINTCNF2.bit.WRBUFFER = 0;
    XintfRegs.XINTCNF2.bit.CLKMODE = 0;
    XintfRegs.XINTCNF2.bit.CLKOFF = 1;
    
    // 配置Zone0
    // XRDLEAD: 2 SYSCLKOUT
    // XRD1: 1 SYSCLKOUT
    // XRD2: 1 SYSCLKOUT
    // XWRLEAD: 2 SYSCLKOUT
    // XWR1: 1 SYSCLKOUT
    // XWR2: 1 SYSCLKOUT
    XintfRegs.XTIMCNT0.all = 0x00002211;
    
    // Zone6配置
    XintfRegs.XTIMCNT6.all = 0x00003322;
    XintfRegs.Z6ADDR.all = 0x100000;
    XintfRegs.Z6CSCTL.all = 0x00000001;
    
    // Zone7配置
    XintfRegs.XTIMCNT7.all = 0x00003322;
    XintfRegs.Z7ADDR.all = 0x200000;
    XintfRegs.Z7CSCTL.all = 0x00000001;
    
    // 切换到XINTF时钟
    asm(" OR #0x0010, 0x26");
}

// 读取XINTF区域数据
uint16_t XintfRead(uint32_t addr)
{
    return *(uint16_t *)addr;
}

// 写入XINTF区域数据
void XintfWrite(uint32_t addr, uint16_t data)
{
    *(uint16_t *)addr = data;
}
