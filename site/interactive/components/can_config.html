// CAN初始化配置
void InitCAN(void)
{
    // 禁用CAN模块进行配置
    CanaRegs.CANCTL.all = 0x00000001;
    
    // 配置位定时器 (BTC)
    // BRP = 4, TSEG1 = 13, TSEG2 = 2
    // 波特率 = 150MHz / (4+1) / (13+2+3) = 1Mbps
    CanaRegs.CANBTC.all = 0x00000000;
    CanaRegs.CANBTC.bit.BRP = 4;
    CanaRegs.CANBTC.bit.TSEG1 = 13;
    CanaRegs.CANBTC.bit.TSEG2 = 2;
    CanaRegs.CANBTC.bit.SAM = 1;  // 采样3次
    
    // 配置消息对象
    // 设置局部屏蔽寄存器
    CanaRegs.CANLAM.all = 0x00000000;
    
    // 配置发送消息对象 1
    CanaRegs.CANMDER.all = 0x00000000;
    CanaRegs.CANTRS.all = 0x00000000;
    
    // 选择消息对象1
    // MSG1 CTRL: DLC = 8, IDE = 0 (标准帧)
    CanaRegs.CANMC.all = 0x00000000;
    CanaRegs.CANMC.bit.CANCCR = 1;
    
    // 等待配置完成
    while(CanaRegs.CANMC.bit.CANCCR);
    
    // MSG1 Arbitration
    // MSG1 MSGID = 0x200
    CanaRegs.CANMSGID.all = 0x2000000;
    
    // MSG1 Control
    // DLC = 8, MSGVAL = 1, TXIE = 1, RXIE = 1
    CanaRegs.CANMSGCTRL.all = 0x00000000;
    CanaRegs.CANMSGCTRL.bit.DLC = 8;
    CanaRegs.CANMSGCTRL.bit.TXRQ = 0;
    
    // 释放CAN模块
    CanaRegs.CANMC.all = 0x00000000;
    CanaRegs.CANCTL.all = 0x00000000;
}

// 发送CAN消息
void CANSend(uint32_t id, uint8_t *data, uint8_t dlc)
{
    // 选择消息对象
    CanaRegs.CANMC.bit.CANCCR = 1;
    while(CanaRegs.CANMC.bit.CANCCR);
    
    // 设置ID
    CanaRegs.CANMSGID.all = id << 18;
    
    // 设置数据
    if(dlc > 0) CanaRegs.CANMDL.all = *(uint32_t *)data;
    if(dlc > 4) CanaRegs.CANMDH.all = *(uint32_t *)(data + 4);
    
    // 设置控制
    CanaRegs.CANMSGCTRL.bit.DLC = dlc;
    
    // 发送请求
    CanaRegs.CANTRS.all = 0x00000001;
    while(!CanaRegs.CANTA.all & 0x00000001);
    
    // 清除应答
    CanaRegs.CANTA.all = 0x00000001;
    
    CanaRegs.CANMC.all = 0x00000000;
}
