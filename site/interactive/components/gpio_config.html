寄存器配置代码

// GPIO初始化配置
void InitGpio(void)
{
    // 将所有GPIO配置为GPIO功能（而非外设功能）
    EALLOW;
    
    // 配置GPIO0-GPIO31为通用GPIO
    GpioCtrlRegs.GPAMUX1.all = 0x00000000;  // GPIO功能
    GpioCtrlRegs.GPAMUX2.all = 0x00000000;  // GPIO功能
    
    // 配置方向：GPIO0-GPIO5为输出（LED），其余为输入
    GpioCtrlRegs.GPADIR.all = 0x0000003F;   // 位0-5输出，位6-31输入
    
    // 禁止上拉
    GpioCtrlRegs.GPAPUD.all = 0x00000000;   // 所有引脚上拉使能
    
    // 配置输入量化
    GpioCtrlRegs.GPAQSEL1.all = 0x00000000; // SYSCLKOUT同步
    GpioCtrlRegs.GPAQSEL2.all = 0x00000000; // SYSCLKOUT同步
    
    EDIS;
}
